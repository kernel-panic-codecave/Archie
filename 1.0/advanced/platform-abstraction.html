    Platform Abstraction¶

    Archie uses Architectury for cross-platform mod development.

    Architecture Overview¶

    Project/
├── common/           # Platform-agnostic code
├── fabric/          # Fabric-specific code
└── neoforge/        # NeoForge-specific code

    Common Module¶

    Code that runs on both platforms:

    // common/src/main/kotlin/com/example/MyMod.kt
object MyMod {
    const val MOD_ID = "mymod"

    @JvmStatic
    fun init() {
        // This runs on both platforms
        MyConfig.init()
        registerContent()
    }
}

    Platform-Specific Entry Points¶

    Fabric¶

    // fabric/src/main/kotlin/com/example/fabric/MyModFabric.kt
package com.example.fabric

import com.example.MyMod
import net.fabricmc.api.ModInitializer

class MyModFabric : ModInitializer {
    override fun onInitialize() {
        MyMod.init()
    }
}

    // fabric/src/main/resources/fabric.mod.json
{
    "schemaVersion": 1,
    "id": "mymod",
    "version": "${version}",
    "entrypoints": {
        "main": [
            "com.example.fabric.MyModFabric"
        ]
    }
}

    NeoForge¶

    // neoforge/src/main/kotlin/com/example/neoforge/MyModNeoForge.kt
package com.example.neoforge

import com.example.MyMod
import net.neoforged.fml.common.Mod
import net.neoforged.fml.event.lifecycle.FMLCommonSetupEvent
import thedarkcolour.kotlinforforge.neoforge.forge.MOD_BUS

@Mod(MyMod.MOD_ID)
class MyModNeoForge {
    init {
        MOD_BUS.addListener<FMLCommonSetupEvent> {
            MyMod.init()
        }
    }
}

    Runtime Detection¶

    import net.kernelpanicsoft.archie.APlatform

when (APlatform.platform) {
    "fabric" -> println("Running on Fabric")
    "neoforge" -> println("Running on NeoForge")
}

    Compile-Time Separation¶

    Use separate source sets for truly platform-specific code:

    // Only in fabric source set
import net.fabricmc.fabric.api.event.lifecycle.v1.ServerLifecycleEvents

// Only in neoforge source set  
import net.neoforged.neoforge.event.server.ServerStartingEvent

    Platform-Specific Implementations¶

    Expect/Actual Pattern¶

    // common/
expect object PlatformHelper {
    fun isModLoaded(modId: String): Boolean
}

// fabric/
actual object PlatformHelper {
    actual fun isModLoaded(modId: String): Boolean {
        return FabricLoader.getInstance().isModLoaded(modId)
    }
}

// neoforge/
actual object PlatformHelper {
    actual fun isModLoaded(modId: String): Boolean {
        return ModList.get().isLoaded(modId)
    }
}

    Conditional Compilation¶

    Using expect/actual¶

    // common/
expect fun registerClientEvents()

// fabric/
actual fun registerClientEvents() {
    ClientTickEvents.END_CLIENT_TICK.register { client ->
        // Fabric client tick
    }
}

// neoforge/
actual fun registerClientEvents() {
    NeoForge.EVENT_BUS.addListener<ClientTickEvent> { event ->
        // NeoForge client tick
    }
}

    Resource Location Conventions¶

    Use consistent resource locations across platforms:

    object MyMod {
    @JvmStatic
    operator fun get(path: String): ResourceLocation =
        ResourceLocation.fromNamespaceAndPath(MOD_ID, path)
}

// Usage
val texture = MyMod["textures/item/my_item.png"]
val blockId = MyMod["my_block"]

    Common Pitfalls¶

    Classpath Issues¶

    // Don't do this in common
import net.fabricmc.api.EnvType  // Fabric-only!

// Do this instead
import net.kernelpanicsoft.archie.APlatform
if (APlatform.isClient) {
    // Client code
}

    Event Differences¶

    Events have different naming between platforms:

    Concept Fabric NeoForge
    Server starting ServerLifecycleEvents.SERVER_STARTING ServerStartingEvent
    Player join ServerPlayConnectionEvents.JOIN PlayerLoggedInEvent
    Tick ServerTickEvents TickEvent.Server

    Registration Differences¶

    // Fabric - uses events
Registry.register(BuiltInRegistries.BLOCK, MyMod["my_block"], MY_BLOCK)

// NeoForge - uses DeferredRegister
object MyBlocks {
    val MY_BLOCK: DeferredRegister<Block> = DeferredRegister.create(BuiltInRegistries.BLOCK, MyMod.MOD_ID)

    val BLOCK: RegistryObject<Block> = MY_BLOCK.register("my_block") { MyBlock() }
}

    Best Practices¶

    1. Maximize common code - Keep as much as possible in common
    2. Use Archie abstractions - Leverage platform-agnostic utilities
    3. Consistent naming - Use same names for equivalent concepts
    4. Test both platforms - CI should build for both
    5. Document differences - Note platform-specific behavior

    Build Configuration¶

    // build.gradle.kts (root)
subprojects {
    apply(plugin = "dev.architectury.loom")

    dependencies {
        "minecraft"(libs.minecraft)
        "mappings"(loom.layered {
            officialMojangMappings()
            parchment(libs.parchment)
        })
    }
}
